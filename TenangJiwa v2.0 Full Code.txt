  async function toggleWakeLock(forceOn = false) {
    const btn = document.getElementById('wakeLockBtn');
    if (!('wakeLock' in navigator)) {
      if (!forceOn) showToast('Wake Lock not supported on this browser');
      return;
    }
    if (state.wakeLock && !forceOn) {
      await state.wakeLock.release();
      state.wakeLock = null;
      btn.classList.remove('on');
      btn.innerHTML = `☀️ <span>${t('wake_off')}</span>`;
      showToast(t('toast_wake_off'));
      return;
    }
    if (!state.wakeLock) {
      try {
        state.wakeLock = await navigator.wakeLock.request('screen');
        btn.classList.add('on');
        btn.innerHTML = `☀️ <span>${t('wake_on')}</span>`;
        if (!forceOn) showToast(t('toast_wake_on'));
        state.wakeLock.addEventListener('release', () => {
          state.wakeLock = null;
          btn.classList.remove('on');
          btn.innerHTML = `☀️ <span>${t('wake_off')}</span>`;
        });
      } catch (e) {}
    }
  }

  /* =========================================================
     7. WEB AUDIO API ENGINE (SOUNDSCAPE + BINAURAL + CHIME)
     ========================================================= */
  const SoundEngine = {
    ctx: null,
    masterGain: null,
    nodes: {},
    humActive: false,
    binauralMode: 'off',

    init() {
      if (!this.ctx) {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        this.ctx = new AudioCtx();
        this.masterGain = this.ctx.createGain();
        this.masterGain.gain.value = 1;
        this.masterGain.connect(this.ctx.destination);
      }
      if (this.ctx.state === 'suspended') this.ctx.resume();
    },

    playSoftChime(freq = 528) {
      if (!state.chimeOn) return;
      this.init();
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
      gain.gain.setValueAtTime(0.001, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.11, this.ctx.currentTime + 0.08);
      gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + 1.4);
      osc.connect(gain);
      gain.connect(this.ctx.destination);
      osc.start();
      osc.stop(this.ctx.currentTime + 1.5);
    },

    createNoiseBuffer(type) {
      const bufferSize = this.ctx.sampleRate * 3;
      const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
      const output = buffer.getChannelData(0);
      let lastOut = 0.0;
      for (let i = 0; i < bufferSize; i++) {
        const white = Math.random() * 2 - 1;
        if (type === 'brown') {
          output[i] = (lastOut + (0.02 * white)) / 1.02;
          lastOut = output[i];
          output[i] *= 3.2;
        } else {
          output[i] = white * 0.5;
        }
      }
      return buffer;
    },

    setChannel(name, volPercent) {
      this.init();
      const vol = volPercent / 100;
      if (vol <= 0) {
        if (this.nodes[name]) this.nodes[name].gain.gain.setTargetAtTime(0, this.ctx.currentTime, 0.15);
        this.updateStatus();
        return;
      }
      if (!this.nodes[name]) {
        const src = this.ctx.createBufferSource();
        src.buffer = this.createNoiseBuffer(name === 'brown' ? 'brown' : 'white');
        src.loop = true;
        const filter = this.ctx.createBiquadFilter();
        const gain = this.ctx.createGain();
        gain.gain.value = 0;

        if (name === 'rain') {
          filter.type = 'bandpass'; filter.frequency.value = 950; filter.Q.value = 0.6;
        } else if (name === 'brown') {
          filter.type = 'lowpass'; filter.frequency.value = 320;
        } else if (name === 'wind') {
          filter.type = 'lowpass'; filter.frequency.value = 450;
          const lfo = this.ctx.createOscillator();
          const lfoGain = this.ctx.createGain();
          lfo.frequency.value = 0.12; lfoGain.gain.value = 260;
          lfo.connect(lfoGain); lfoGain.connect(filter.frequency);
          lfo.start();
        }
        src.connect(filter); filter.connect(gain); gain.connect(this.masterGain);
        src.start();
        this.nodes[name] = { src, gain };
      }
      const scale = name === 'rain' ? 0.35 : (name === 'brown' ? 0.55 : 0.45);
      this.nodes[name].gain.gain.setTargetAtTime(vol * scale, this.ctx.currentTime, 0.15);
      this.updateStatus();
    },

    setBinaural(mode) {
      this.init();
      // Stop existing binaural
      if (this.nodes.binaural) {
        try {
          this.nodes.binaural.oscL.stop();
          this.nodes.binaural.oscR.stop();
        } catch(e) {}
        delete this.nodes.binaural;
      }
      this.binauralMode = mode;
      const st = document.getElementById('binauralStatus');

      if (mode === 'off') {
        st.textContent = 'OFF';
        return;
      }

      const baseFreq = mode === 'delta' ? 140 : 180;
      const beatDiff = mode === 'delta' ? 2 : 6; // 2Hz Delta / 6Hz Theta
      const oscL = this.ctx.createOscillator();
      const oscR = this.ctx.createOscillator();
      const gain = this.ctx.createGain();

      oscL.type = 'sine'; oscL.frequency.value = baseFreq;
      oscR.type = 'sine'; oscR.frequency.value = baseFreq + beatDiff;

      const vol = (+document.getElementById('volBinaural').value / 100) * 0.18;
      gain.gain.value = vol;

      if (this.ctx.createStereoPanner) {
        const panL = this.ctx.createStereoPanner(); panL.pan.value = -1;
        const panR = this.ctx.createStereoPanner(); panR.pan.value = 1;
        oscL.connect(panL); panL.connect(gain);
        oscR.connect(panR); panR.connect(gain);
      } else {
        oscL.connect(gain); oscR.connect(gain);
      }

      gain.connect(this.masterGain);
      oscL.start(); oscR.start();
      this.nodes.binaural = { oscL, oscR, gain };
      st.textContent = mode === 'delta' ? 'DELTA 2Hz' : 'THETA 6Hz';
    },

    setBinauralVol(percent) {
      if (this.nodes.binaural && this.ctx) {
        this.nodes.binaural.gain.gain.setTargetAtTime((percent / 100) * 0.18, this.ctx.currentTime, 0.1);
      }
    },

    toggleFocusHum() {
      this.init();
      const btn = document.getElementById('toggleFocusHumBtn');
      const st = document.getElementById('focusHumStatus');
      if (!this.humActive) {
        const osc1 = this.ctx.createOscillator();
        const osc2 = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc1.type = 'sine'; osc1.frequency.value = 136.1;
        osc2.type = 'sine'; osc2.frequency.value = 140.1;
        gain.gain.value = 0.08;
        osc1.connect(gain); osc2.connect(gain); gain.connect(this.ctx.destination);
        osc1.start(); osc2.start();
        this.nodes.hum = { osc1, osc2, gain };
        this.humActive = true;
        btn.textContent = t('stop_hum');
        st.textContent = 'ON';
      } else {
        if (this.nodes.hum) {
          this.nodes.hum.gain.gain.setTargetAtTime(0, this.ctx.currentTime, 0.2);
          setTimeout(() => {
            try { this.nodes.hum.osc1.stop(); this.nodes.hum.osc2.stop(); } catch(e){}
            delete this.nodes.hum;
          }, 300);
        }
        this.humActive = false;
        btn.textContent = t('toggle_hum');
        st.textContent = 'OFF';
      }
    },

    stopAllMixer() {
      ['rain', 'brown', 'wind'].forEach(k => {
        if (this.nodes[k]) this.nodes[k].gain.gain.setTargetAtTime(0, this.ctx.currentTime, 0.15);
      });
      ['volRain', 'volBrown', 'volWind'].forEach(id => document.getElementById(id).value = 0);
      ['valRain', 'valBrown', 'valWind'].forEach(id => document.getElementById(id).textContent = '0%');
      this.setBinaural('off');
      ['binOff', 'binDelta', 'binTheta'].forEach(id => document.getElementById(id).classList.remove('active'));
      document.getElementById('binOff').classList.add('active');
      this.updateStatus();
    },

    fadeOutMaster(seconds = 6) {
      HypnoVoice.stop();
      if (!this.ctx || !this.masterGain) return;
      this.masterGain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + seconds);
      setTimeout(() => {
        this.stopAllMixer();
        this.masterGain.gain.value = 1;
      }, seconds * 1000);
    },

    updateStatus() {
      const active = (+volRain.value > 0 || +volBrown.value > 0 || +volWind.value > 0);
      document.getElementById('audioStatus').textContent = active ? 'PLAYING' : 'OFF';
    }
  };

  /* =========================================================
     8. MINDIFI SOFT FEMALE GUIDED VOICE (WEB SPEECH API)

  const HypnoVoice = {
    scriptKey: 'deep_sleep',
    active: false,
    paused: false,
    index: 0,
    pauseTimer: null,
    selectedVoice: null,

    pickFemaleVoice() {
      if (!('speechSynthesis' in window)) return;
      const voices = window.speechSynthesis.getVoices();
      if (!voices.length) return;

      const langPrefix = state.lang === 'ms' ? ['ms', 'id'] : ['en-gb', 'en-us', 'en'];
      const femaleHints = ['female', 'samantha', 'victoria', 'karen', 'moira', 'tessa', 'zira', 'aria', 'jenny', 'yasmin', 'gadis', 'google', 'siti'];

      let chosen = null;
      for (const lp of langPrefix) {
        const langVoices = voices.filter(v => v.lang.toLowerCase().startsWith(lp));
        if (langVoices.length) {
          chosen = langVoices.find(v => femaleHints.some(h => v.name.toLowerCase().includes(h))) || langVoices[0];
          break;
        }
      }
      this.selectedVoice = chosen || voices[0];
      const vName = this.selectedVoice ? this.selectedVoice.name.slice(0, 26) : 'Default Soft';
      document.getElementById('activeVoiceName').textContent = `Voice: ${vName}`;
    },

    startOrToggle() {
      if (!('speechSynthesis' in window)) {
        showToast('Speech API not supported on this device');
        return;
      }
      if (!this.active) {
        this.active = true;
        this.paused = false;
        this.index = 0;
        toggleWakeLock(true);
        document.getElementById('hypnoState').textContent = 'SPEAKING';
        document.getElementById('hypnoPlayBtn').textContent = t('pause_hypno');
        this.speakCurrent();
      } else if (!this.paused) {
        this.paused = true;
        clearTimeout(this.pauseTimer);
        window.speechSynthesis.cancel();
        document.getElementById('hypnoState').textContent = 'PAUSED';
        document.getElementById('hypnoPlayBtn').textContent = t('resume_hypno');
      } else {
        this.paused = false;
        document.getElementById('hypnoState').textContent = 'SPEAKING';
        document.getElementById('hypnoPlayBtn').textContent = t('pause_hypno');
        this.speakCurrent();
      }
    },

    speakCurrent() {
      if (!this.active || this.paused) return;
      const lines = HYPNO_SCRIPTS[this.scriptKey][state.lang];
      if (this.index >= lines.length) {
        this.stop();
        return;
      }
      const line = lines[this.index];
      document.getElementById('hypnoSubtitle').textContent = `"${line}"`;
      document.getElementById('hypnoStepInfo').textContent = `${this.index + 1} / ${lines.length}`;
      document.getElementById('hypnoBar').style.width = `${((this.index + 1) / lines.length) * 100}%`;

      window.speechSynthesis.cancel();
      const utter = new SpeechSynthesisUtterance(line);
      if (!this.selectedVoice) this.pickFemaleVoice();
      if (this.selectedVoice) utter.voice = this.selectedVoice;
      utter.lang = state.lang === 'ms' ? 'ms-MY' : 'en-GB';
      utter.rate = 0.78;  // Slow Mindifi hypnotic cadence
      utter.pitch = 0.96; // Soft warm female tone
      utter.volume = (+document.getElementById('volVoice').value) / 100;

      utter.onend = () => {
        if (!this.active || this.paused) return;
        // 3.2 seconds breathing pause between hypnotic lines
        this.pauseTimer = setTimeout(() => {
          this.index++;
          this.speakCurrent();
        }, 3200);
      };

      window.speechSynthesis.speak(utter);
    },

    stop() {
      this.active = false;
      this.paused = false;
      this.index = 0;
      clearTimeout(this.pauseTimer);
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      document.getElementById('hypnoState').textContent = 'READY';
      document.getElementById('hypnoPlayBtn').textContent = t('start_hypno');
      document.getElementById('hypnoSubtitle').textContent = t('hypno_idle');
      document.getElementById('hypnoBar').style.width = '0%';
      const total = HYPNO_SCRIPTS[this.scriptKey][state.lang].length;
      document.getElementById('hypnoStepInfo').textContent = `0 / ${total}`;
    }
  };

  function startBreathing() {
    breathRunning = true;
    toggleWakeLock(true);
    document.getElementById('breathToggleBtn').textContent = t('stop_breath');
    const steps = breathMode === 'box'
      ? [{ k: 'inhale', s: 4, sc: 1.48, f: 528 }, { k: 'hold', s: 4, sc: 1.48, f: 440 }, { k: 'exhale', s: 4, sc: 1.0, f: 396 }, { k: 'hold', s: 4, sc: 1.0, f: 440 }]
      : [{ k: 'inhale', s: 4, sc: 1.52, f: 528 }, { k: 'hold', s: 7, sc: 1.52, f: 440 }, { k: 'exhale', s: 8, sc: 1.0, f: 396 }];

    let idx = 0;
    const ring = document.getElementById('breathRing');
    const lbl = document.getElementById('breathLabel');
    const cnt = document.getElementById('breathCount');

    function runStep() {
      if (!breathRunning) return;
      const st = steps[idx];
      ring.style.transition = `transform ${st.s}s ease-in-out`;
      ring.style.transform = `scale(${st.sc})`;
      lbl.textContent = t(st.k);

      // Eyes-closed guidance: soft chime + gentle vibration
      SoundEngine.playSoftChime(st.f);
      vibrate(st.k === 'inhale' ? [60] : (st.k === 'exhale' ? [40, 40, 40] : [25]));

      let rem = st.s;
      cnt.textContent = rem;
      clearInterval(breathTick);
      breathTick = setInterval(() => {
        rem--;
        if (rem > 0) cnt.textContent = rem;
      }, 1000);

      breathTimeout = setTimeout(() => {
        idx = (idx + 1) % steps.length;
        runStep();
      }, st.s * 1000);
    }
    runStep();
  }

  function stopBreathing() {
    breathRunning = false;
    clearTimeout(breathTimeout);
    clearInterval(breathTick);
    const ring = document.getElementById('breathRing');
    ring.style.transition = 'transform 0.5s ease';
    ring.style.transform = 'scale(1)';
    document.getElementById('breathLabel').textContent = t('ready');
    document.getElementById('breathCount').textContent = '•';
    document.getElementById('breathToggleBtn').textContent = t('start_breath');
  }

  /* =========================================================
     10. SLEEP TIMER WITH CANCEL
     ========================================================= */
  let sleepInterval = null;
  function startSleepTimer(mins, isProFeature = false) {
    if (isProFeature && !state.isPro) {
      showToast(t('toast_pro_req'));
      switchTab('pro');
      return;
    }
    clearInterval(sleepInterval);
    let sec = mins * 60;
    const disp = document.getElementById('sleepTimerDisplay');
    document.getElementById('cancelSleepTimerBtn').style.display = 'block';
    showToast(`Sleep Timer: ${mins} min`);

    sleepInterval = setInterval(() => {
      sec--;
      const m = String(Math.floor(sec / 60)).padStart(2, '0');
      const s = String(sec % 60).padStart(2, '0');
      disp.textContent = `${m}:${s}`;
      if (sec === 7) SoundEngine.fadeOutMaster(6);
      if (sec <= 0) cancelSleepTimer(true);
    }, 1000);
  }

  function cancelSleepTimer(silent = false) {
    clearInterval(sleepInterval);
    sleepInterval = null;
    document.getElementById('sleepTimerDisplay').textContent = '--:--';
    document.getElementById('cancelSleepTimerBtn').style.display = 'none';
    if (!silent) showToast(t('toast_timer_cancel'));
  }

  /* =========================================================
     11. FOCUS TIMER
     ========================================================= */
  let focusTotal = 25 * 60;
  let focusLeft = focusTotal;
  let focusInterval = null;

  function renderFocus() {
    const m = String(Math.floor(focusLeft / 60)).padStart(2, '0');
    const s = String(focusLeft % 60).padStart(2, '0');
    document.getElementById('focusDisplay').textContent = `${m}:${s}`;
  }

  function setFocusDuration(mins, isProFeature, btnEl) {
    if (isProFeature && !state.isPro) {
      showToast(t('toast_pro_req'));
      switchTab('pro');
      return;
    }
    clearInterval(focusInterval);
    focusInterval = null;
    document.getElementById('focusStartBtn').textContent = t('start_focus');
    document.querySelectorAll('#view-focus .pill').forEach(p => p.classList.remove('active'));
    btnEl.classList.add('active');
    focusTotal = mins * 60;
    focusLeft = focusTotal;
    renderFocus();
  }

  /* =========================================================
     12. SOS PANIC MODAL
     ========================================================= */
  let sosInt = null;
  let sosPhaseInt = null;
  function openSos() {
    const modal = document.getElementById('sosModal');
    const circle = document.getElementById('sosCircle');
    const cue = document.getElementById('sosCue');
    const timer = document.getElementById('sosTimer');
    modal.classList.add('open');

    let left = 60;
    timer.textContent = left + 's';
    let inhale = true;

    function pulse() {
      if (inhale) {
        cue.textContent = t('inhale');
        circle.style.transform = 'scale(1.28)';
        SoundEngine.playSoftChime(528);
        vibrate([60]);
      } else {
        cue.textContent = t('exhale');
        circle.style.transform = 'scale(0.92)';
        SoundEngine.playSoftChime(396);
        vibrate([40, 40]);
      }
      inhale = !inhale;
    }
    pulse();
    sosPhaseInt = setInterval(pulse, 4000);
    sosInt = setInterval(() => {
      left--;
      timer.textContent = left + 's';
      if (left <= 0) closeSos();
    }, 1000);
  }

  function closeSos() {
    clearInterval(sosInt);
    clearInterval(sosPhaseInt);
    document.getElementById('sosModal').classList.remove('open');
  }
